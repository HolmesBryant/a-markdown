```python
# Some python
#!/usr/bin/env python3

def get_item_cost(name):
	costs = {"sword": 50, "potion": 10}
	if name in costs: return costs[name]
	else: return 0

result = get_item_cost("sword")
print(result)
```

```php
// Some php
function calculateDamage($type, $level) {
	if ($type === 'fire') { return $level * 2; }
	elseif ($type === 'ice') { return $level * 3; }
	else { return $level; }
}

echo calculateDamage('fire', 5);
```
